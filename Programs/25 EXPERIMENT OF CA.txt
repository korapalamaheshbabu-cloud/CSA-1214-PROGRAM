.MODEL SMALL
.STACK 100H

.DATA
    NUM1 DW 5678H
    NUM2 DW 1234H
    DIFF DW ?

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

    MOV AX, NUM1      ; Load first 16-bit number
    SUB AX, NUM2      ; Subtract second 16-bit number
    MOV DIFF, AX      ; Store the result

    MOV AH, 4CH       ; Exit to DOS
    INT 21H

MAIN ENDP
END MAIN
