.MODEL SMALL
.STACK 100H

.DATA
    NUM1 DW 1234H
    NUM2 DW 5678H
    SUM  DW ?

.CODE
MAIN PROC
    MOV AX, @DATA
    MOV DS, AX

    MOV AX, NUM1      ; Load first 16-bit number into AX
    ADD AX, NUM2      ; Add second 16-bit number
    MOV SUM, AX       ; Store the result

    MOV AH, 4CH       ; Exit to DOS
    INT 21H

MAIN ENDP
END MAIN
